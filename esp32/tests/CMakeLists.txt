cmake_minimum_required(VERSION 3.16)
project(push_host_tests C CXX)
if(NOT EXISTS "${MBEDTLS_SOURCE}/include/mbedtls/hkdf.h")
  message(FATAL_ERROR "Pass -D MBEDTLS_SOURCE=<IDF 5.5.5>/components/mbedtls/mbedtls")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
add_executable(push_driver driver.cpp ../main/push_crypto.cpp ../main/policy.cpp)
target_include_directories(push_driver PRIVATE ../main)
target_compile_definitions(push_driver PRIVATE WP_TEST)
target_compile_options(push_driver PRIVATE -Wall -Wextra -Werror)
target_link_libraries(push_driver PRIVATE mbedcrypto)
# Compile the actual serial/event adapter against host-only SDK stand-ins.
set(stub_headers freertos/FreeRTOS.h freertos/event_groups.h freertos/task.h
  lwip/sockets.h esp_event.h esp_wifi.h esp_netif.h esp_sntp.h esp_random.h esp_timer.h
  esp_heap_caps.h esp_http_client.h esp_crt_bundle.h esp_log.h nvs_flash.h poc_config.h)
foreach(header IN LISTS stub_headers)
  get_filename_component(directory "${CMAKE_CURRENT_BINARY_DIR}/stubs/${header}" DIRECTORY)
  file(MAKE_DIRECTORY "${directory}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/stubs/${header}" "#include \"sdk.h\"\n")
endforeach()
add_executable(serial_tests serial.cpp ../main/policy.cpp ../main/push_crypto.cpp)
target_include_directories(serial_tests PRIVATE stubs "${CMAKE_CURRENT_BINARY_DIR}/stubs" ../main)
target_compile_features(serial_tests PRIVATE cxx_std_17)
target_compile_options(serial_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(serial_tests PRIVATE mbedcrypto)
