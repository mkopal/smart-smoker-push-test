cmake_minimum_required(VERSION 3.16)
project(push_host_tests C CXX)
if(NOT EXISTS "${MBEDTLS_SOURCE}/include/mbedtls/hkdf.h")
  message(FATAL_ERROR "Pass -D MBEDTLS_SOURCE=<IDF 5.5.5>/components/mbedtls/mbedtls")
endif()
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${MBEDTLS_SOURCE}" mbedtls)
add_executable(push_driver driver.cpp ../main/push_crypto.cpp ../main/policy.cpp)
target_include_directories(push_driver PRIVATE ../main)
target_compile_definitions(push_driver PRIVATE WP_TEST)
target_compile_options(push_driver PRIVATE -Wall -Wextra -Werror)
target_link_libraries(push_driver PRIVATE mbedcrypto)
